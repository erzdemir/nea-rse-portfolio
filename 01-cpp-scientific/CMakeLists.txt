cmake_minimum_required(VERSION 3.16)
project(nea_xs_toolkit LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
add_library(nea_xs src/xs_table.cpp)
target_include_directories(nea_xs PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
add_executable(xs_lookup src/main.cpp)
add_executable(xs_compare src/app_compare.cpp)
add_executable(test_xs_table tests/test_xs_table.cpp)
target_link_libraries(xs_lookup PRIVATE nea_xs)
target_link_libraries(xs_compare PRIVATE nea_xs)
target_link_libraries(test_xs_table PRIVATE nea_xs)
enable_testing()
add_test(NAME xs_table COMMAND test_xs_table)
